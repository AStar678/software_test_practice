## Reproduction (backend/ as cwd; project MySQL already running on 127.0.0.1:13306)

```bash
mvn -B -ntp -o -Dmaven.repo.local=.runtime/m2 -DskipTests package
python3 scripts/check_web_resources.py --repo .. --jar target/pandora-backend-0.1.0.jar \
  --output ../audit/2026-10-27/chapter10-web-resources.json

# dedicated isolated demo DB, same env as run_check_isolated.sh
set -a; source .runtime/database.env; set +a
DB="pandora_demo_e13_$(date +%s)"
mysql --defaults-file=.runtime/mysql-admin.cnf -e "CREATE DATABASE \`$DB\` CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci;
  GRANT SELECT,INSERT,UPDATE,DELETE,CREATE,ALTER,INDEX,REFERENCES,DROP ON \`$DB\`.* TO '$PANDORA_DB_USER'@'localhost';
  GRANT SELECT,INSERT,UPDATE,DELETE,CREATE,ALTER,INDEX,REFERENCES,DROP ON \`$DB\`.* TO '$PANDORA_DB_USER'@'127.0.0.1'; FLUSH PRIVILEGES;"
export PANDORA_DB_URL="jdbc:mysql://127.0.0.1:13306/$DB?connectionTimeZone=UTC&forceConnectionTimeZoneToSession=true&useUnicode=true&characterEncoding=UTF-8&allowPublicKeyRetrieval=true&sslMode=DISABLED"
export PANDORA_SEED=true PANDORA_WORKER_ENABLED=true PANDORA_PORT=18130 PANDORA_BIND_ADDRESS=127.0.0.1
java -jar target/pandora-backend-0.1.0.jar --pandora.seed=true --pandora.worker-enabled=true &   # wait for /health and seeded users

python3 scripts/prepare_demo.py --base-url http://127.0.0.1:18130 --output ../audit/2026-10-27/chapter11-demo-prep-1.json
python3 scripts/prepare_demo.py --base-url http://127.0.0.1:18130 --output ../audit/2026-10-27/chapter11-demo-prep-2.json
node scripts/check_demo_flow_browser.js --base-url http://127.0.0.1:18130 \
  --output ../audit/2026-10-27/chapter11-demo-web.json
# stop the backend, then query the counts shown in chapter11-demo-db.txt
# restart the backend on the SAME database, then:
node scripts/check_demo_home_style.js --base-url http://127.0.0.1:18130 \
  --output ../audit/2026-10-27/chapter11-demo-home-style.json \
  --screenshot ../audit/2026-10-27/chapter11-demo-home.png
# stop the backend and DROP DATABASE `$DB`
```
